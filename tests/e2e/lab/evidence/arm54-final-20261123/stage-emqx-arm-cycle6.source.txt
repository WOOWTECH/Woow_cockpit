import hashlib, json, os, pathlib
root = pathlib.Path('/var/tmp/woow-arm54-final-20261123-wcjdtakj')
def sha(p):
    d = hashlib.sha256()
    with open(p, 'rb') as f:
        for b in iter(lambda: f.read(1 << 20), b''):
            d.update(b)
    return d.hexdigest()
out = {name: sha(root / 'input' / name) == want for name, want in sorted({'chain-b1-arm-cold.json': '17c35c8d0a974fa7403e708704552cfffba39586ec35f26755ef8d242e702241', 'ha-backup-b1-arm-cold.tar': 'b6eb257cc5790bee30500f929d51ade66bfa8354dbe5bb61ba7c4bc08582f664', 'emqx-expect-b1-arm-cold.json': '977dd497142495efd571946e9b9ecfaea7c3dd948fb7ac3d882a85369429ed8c'}.items())}
for name, label in (('ha-key', 'staged_backup_unlock_file_private'), ('n8n-creds.json', 'staged_n8n_login_file_private'),
                    ('emqx-creds.json', 'staged_emqx_login_file_private')):
