import hashlib, json, os, pathlib, tarfile
root = pathlib.Path('/var/tmp/woow-arm54-final-20261123-1md3hvc1'); os.umask(0o077)
def sha(p):
    d = hashlib.sha256()
    with open(p, 'rb') as f:
        for b in iter(lambda: f.read(1 << 20), b''):
            d.update(b)
    return d.hexdigest()
out = {}
for name, want in sorted({'age.deb': 'd14f937ecb1cef9279879e779cf61fda682735dd9871cd7d60ba7a487608f4e6', 'zstd.deb': '97a58383b0ad55cc8183e59cbb112f2eb55c3dc1e90de816fa5cfb983d5c2ff0', 'code.tar.gz': '552505e0496da1431cb6b2e12a672088425ab3d41c424a8247d97863d40ffe64'}.items()):
    out[name] = sha(root / 'input' / name) == want
