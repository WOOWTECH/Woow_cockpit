import hashlib, json, os, pathlib, tarfile
root = pathlib.Path('/var/tmp/woow-arm54-final-20261123-jp0tjai3'); os.umask(0o077)
def sha(p):
    d = hashlib.sha256()
    with open(p, 'rb') as f:
        for b in iter(lambda: f.read(1 << 20), b''):
            d.update(b)
    return d.hexdigest()
out = {}
for name, want in sorted({'code.tar.gz': '93fa6169b1a4fcab8e24358313d8b1fe8436621c1114580029d6959025dbaa49', 'chain.json': 'f1737d422c79fa5a6b9abc53b8a3ae9cc1003f46f06905c8ec5190b3d8b45f34', 'ha-backup.tar': '1e6a4665315bc69fd114f8d43f7d429aecab53f07d07793ecf8c2c057a92c5d4', 'omni-expect.json': '288e21968edb1c540251a10107dcc05e64bd1f60d9f5bbd3f8b5ccd87bdcbf58'}.items()):
    out[name] = sha(root / 'input' / name) == want
