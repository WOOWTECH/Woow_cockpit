import hashlib, json, os, pathlib, tarfile
root = pathlib.Path('/var/tmp/woow-arm54-final-20261123-h3tlmdwu'); os.umask(0o077)
def sha(p):
    d = hashlib.sha256()
    with open(p, 'rb') as f:
        for b in iter(lambda: f.read(1 << 20), b''):
            d.update(b)
    return d.hexdigest()
out = {}
for name, want in sorted({'code.tar.gz': '21742cf3dd25a48b426f9ca182220e792132139811f6229543605f2f17e615e7', 'chain.json': 'a9fd54b5d23ceb7df393476fbb76b7af7a96612794e45010146218d1f9e762aa', 'ha-backup.tar': '546aecf3be5a13642603bd901d49ee4de01313049d4cddf8805ef8c2b973be44'}.items()):
    out[name] = sha(root / 'input' / name) == want
with tarfile.open(root / 'input' / 'code.tar.gz') as tar:
    for m in tar:
        p = pathlib.PurePosixPath(m.name)
        assert m.isfile() and m.size <= 8 * 1024 ** 2 and not p.is_absolute() and '..' not in p.parts
        dest = root / 'source' / m.name
        dest.parent.mkdir(parents=True, exist_ok=True, mode=0o700)
        with dest.open('xb') as handle:
            handle.write(tar.extractfile(m).read())
