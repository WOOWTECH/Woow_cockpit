def private(value):
    if isinstance(value, str):
        value = value.encode()
    if isinstance(value, bytes) and len(value.strip()) >= 6:

def check(name, ok, **detail):

def mem_available():
    for line in Path('/proc/meminfo').read_text().splitlines():
        if line.startswith('MemAvailable:'):
            return int(line.split()[1]) * 1024
    return None

def render(data):
    for key, value in RENDER.items():
        data = data.replace('@@%s@@' % key, value)
    assert '@@' not in data, 'unrendered-placeholder'
    return data


def package_quadlets():
    """Queue mode exactly like app_render: base units + queue units, n8n-main.fragment appended to n8n.container,
    the optional runners unit (N8N_RUNNERS_MODE=external), no webhook processor (N8N_QUEUE_WEBHOOK_PROCESSOR=false)."""
    q = PACKAGE / 'quadlet'
    files = {}
    for path in sorted(q.glob('*.container')) + sorted(q.glob('*.volume')) + sorted(q.glob('*.network')):
        files[path.name] = path.read_text()
    files['n8n-runners.container'] = (q / 'optional/n8n-runners.container').read_text()
    for path in sorted((q / 'queue').glob('*.container')) + sorted((q / 'queue').glob('*.volume')):
        files[path.name] = path.read_text()
    files['n8n.container'] += (q / 'queue/n8n-main.fragment').read_text()
    files.pop('n8n-webhook.container', None)
    return {name: render(body) for name, body in files.items()}


def env_file():
    """~/.config/n8n/n8n.env from the package example with the host's choices (queue mode, runners external)."""
    lines = []
    for line in (PACKAGE / 'config/n8n.env.example').read_text().splitlines():
        if line.startswith('EXECUTIONS_MODE='):
            line = 'EXECUTIONS_MODE=queue'
        lines.append(line)
    return '\n'.join(lines) + '\n'


def volume_name(quadlets, source):
    if source.endswith('.volume'):
        return [v for k, v in tempctr.parse_unit(quadlets[source]).get('Volume', []) if k == 'VolumeName'][-1]
    return source


def container_args(quadlets, name):
    parsed = tempctr.parse_unit(quadlets[name]).get('Container', [])
    args, used, command = ['--detach', '--pod', POD, '--label', LABEL, '--pull=never'], set(), []
    container_name = image_ref = None
    for key, value in parsed:
        used.add(key)
        if key == 'ContainerName':
            container_name = value
        elif key == 'Image':
            image_ref = value
        elif key == 'Volume':
            source, _sep, rest = value.partition(':')
            source = source.replace('%h', str(ROOT))
            args += ['--volume', volume_name(quadlets, source) + ':' + rest]
        elif key == 'Environment':
            for item in shlex.split(value):
                args += ['--env', item]
        elif key == 'EnvironmentFile':
            args += ['--env-file', value.lstrip('-').replace('%h', str(ROOT))]
        elif key == 'Secret':
            args += ['--secret', value]
        elif key == 'Exec':
            command = shlex.split(value)
    skipped = sorted(used - {'ContainerName', 'Image', 'Volume', 'Environment', 'EnvironmentFile', 'Secret', 'Exec'})
    if container_name == MAIN:
        # Stated deviation: one pod network namespace instead of the n8n bridge network; the package's host names
        # n8n-postgres / n8n-valkey are the pod's loopback (later --env wins over the quadlet's Environment=).
        args += ['--env', 'DB_POSTGRESDB_HOST=127.0.0.1', '--env', 'QUEUE_BULL_REDIS_HOST=127.0.0.1']
    args += ['--name', container_name, '--memory', MEMORY[container_name], '--pids-limit', '1024']
    return container_name, args + [image_ref, *command], image_ref, skipped


# ---------------------------------------------------------------- HTTP in the pod's namespace

class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, *args, **kwargs):
        return None


OPENER = urllib.request.build_opener(urllib.request.ProxyHandler({}), NoRedirect())


def http(method, path, body=None, cookie=None, timeout=30):
    data = json.dumps(body).encode() if body is not None else None
    headers = {'Content-Type': 'application/json', 'Accept': 'application/json', 'browser-id': BROWSER_ID}
    if cookie:
        headers['Cookie'] = 'n8n-auth=' + cookie
    request = urllib.request.Request(BASE_URL + path, data=data, headers=headers, method=method)
    try:
        with OPENER.open(request, timeout=timeout) as response:
            raw = response.read(8 * 1024 * 1024 + 1)
            assert len(raw) <= 8 * 1024 * 1024, 'response-too-large'
            jar = [h for k, h in response.headers.items() if k.lower() == 'set-cookie']
            try:
                return response.status, (json.loads(raw) if raw else None), jar
            except ValueError:
                return response.status, None, jar
    except urllib.error.HTTPError as error:
        return error.code, None, []
    except (urllib.error.URLError, OSError):
        return None, None, []


def wait_ready(seconds=600):
    deadline = time.monotonic() + seconds
    while time.monotonic() < deadline:
        if http('GET', '/healthz/readiness', timeout=10)[0] == 200:
            return True
        sample_mem()
        time.sleep(3)
    raise AssertionError('n8n-not-ready')


def login(email, password):
    status, _body, jar = http('POST', '/rest/login', {'emailOrLdapLoginId': email, 'password': password})
    cookie = None
    for header in jar:
        first = header.split(';', 1)[0]
        if first.startswith('n8n-auth='):
            cookie = first[len('n8n-auth='):]
    if cookie:
        private(cookie)
    return status, cookie


def data_of(body):
    if isinstance(body, dict) and isinstance(body.get('data'), list):
        return body['data']
    if isinstance(body, dict) and isinstance(body.get('data'), dict) and isinstance(body['data'].get('data'), list):
        return body['data']['data']
    return body if isinstance(body, list) else []


def sql(statement):
    return podman('exec', '-i', DB, 'psql', '-X', '-A', '-t', '-v', 'ON_ERROR_STOP=1', '-U', 'n8n', '-d', 'n8n',
                  '-f', '-', input=statement.encode(), timeout=120).stdout.decode().strip()


def setns_pod():
    infra = text('pod', 'inspect', POD, '--format', '{{.InfraContainerID}}').strip()
    pid = int(text('inspect', infra, '--format', '{{.State.Pid}}').strip())
    libc = ctypes.CDLL(None, use_errno=True)
    fd = os.open('/proc/%d/ns/net' % pid, os.O_RDONLY)
    try:
        if libc.setns(fd, CLONE_NEWNET) != 0:
            raise OSError(ctypes.get_errno(), 'setns')
    finally:
        os.close(fd)
    return sorted(name for _index, name in socket.if_nameindex())


def wait_db(seconds=240):
    deadline = time.monotonic() + seconds
    while time.monotonic() < deadline:
        p = podman('exec', DB, 'pg_isready', '-h', '127.0.0.1', '-p', '5432', '-U', 'n8n', '-d', 'n8n',
                   check=False, timeout=20)
        if p.returncode == 0:
            return True
        time.sleep(2)
    raise AssertionError('db-not-ready')


def wait_valkey(seconds=120):
    deadline = time.monotonic() + seconds
    while time.monotonic() < deadline:
        p = podman('exec', VALKEY, '/bin/sh', '/usr/local/bin/start-valkey.sh', 'health', check=False, timeout=20)
        if p.returncode == 0:
            return True
        time.sleep(2)
    raise AssertionError('valkey-not-ready')

def stop_all(remove=True):
    for name in (MAIN, VALKEY, DB):
        podman('stop', '--time', '60', name, check=False, timeout=120)

def export_file(volume, path):
    with open(path, 'xb') as handle:
        os.fchmod(handle.fileno(), 0o600)
        podman('volume', 'export', volume, stdout=handle, timeout=900, max_output=4 * 1024 ** 3)
    return path


def import_file(volume, path):
    with open(path, 'rb') as handle:
        podman('volume', 'import', volume, '-', input=handle, timeout=900)


def tar_member(path, wanted):
    with tarfile.open(path) as tar:
        for member in tar:
            if member.isfile() and member.name.removeprefix('./') == wanted:
                return member.mode & 0o7777, member.uid, tar.extractfile(member).read()
    return None, None, None


def decrypted_credential(name):
    """n8n's own CLI in the running main container (the REST API redacts secrets); the export exists only inside
    the container and is removed; only found/match leave this function."""
    try:
        podman('exec', MAIN, 'n8n', 'export:credentials', '--all', '--decrypted', '--output=/tmp/a54-creds.json',
               timeout=600, max_output=1024 * 1024)
        raw = podman('exec', MAIN, 'cat', '/tmp/a54-creds.json', timeout=60, max_output=4 * 1024 * 1024).stdout
    finally:
        podman('exec', MAIN, 'rm', '-f', '/tmp/a54-creds.json', check=False, timeout=60)
    rows = json.loads(raw or b'[]')
    row = next((x for x in rows if x.get('name') == name), None)
    value = ((row or {}).get('data') or {}).get('value')
    if isinstance(value, str):
        private(value)
    return row is not None, value


def community_nodes(cookie):
    """The moved community package: its database rows and every node type answering /rest/node-types."""
    packages = sql('SELECT "packageName" || \' \' || "installedVersion" FROM installed_packages ORDER BY 1;')
    nodes = [line.split('|') for line in sql('SELECT type, "latestVersion" FROM installed_nodes ORDER BY 1;')
             .splitlines() if line]
    loaded = 0
    for node_type, version in nodes:
        status, body, _jar = http('POST', '/rest/node-types', {'nodeInfos': [{'name': node_type,
                                                                              'version': int(version)}]}, cookie)
        described = data_of(body) if status == 200 else []
        versions = described[0].get('version') if len(described) == 1 else None
        versions = versions if isinstance(versions, list) else [versions]
        if len(described) == 1 and described[0].get('name') in (node_type, node_type.split('.', 1)[-1]) \
                and int(version) in versions:
            loaded += 1
    return packages.splitlines(), len(nodes), loaded

    app = SimpleNamespace(id='n8n', name='n8n 工作流程', owner='arm54-rootful-private-store',
                          volumes=[{'name': DATAVOL}, {'name': VALKEYVOL}], data_dirs=[], secrets=[],
                          postgres=[{'id': 'main', 'container': DB, 'user': 'n8n', 'database': 'n8n',
                                     'scope': 'database', 'unit': 'n8n-postgres.service', 'data_volume': PGVOL}],
                          units=['n8n-postgres.service', 'n8n-valkey.service', 'n8n.service'],
                          health=[{'type': 'http', 'url': 'http://127.0.0.1:15678/healthz/readiness',
                                   'timeout_seconds': 120}])
    stop_all()  # stated: the target is stopped while the product reads the backup and builds the import

    out['phase'] = 'product-check'
    staging = private_dir('work', 'staging')
    adapter = QuadletPostgresAdapter(app, r, LOGS.append)
    temps = {'current': tempctr.TempObjects(r, JOB_CHECK, memory='768m', pids=1024, cpus='1')}
    conversion = temps['current']
    opened = habackup.open_backup(ARCHIVE)
    work = SimpleNamespace(ha=opened, keys=KEYS, _addons={}, staging=staging,
                           guard=haimport.SpaceGuard([staging, ROOT], 12 * 1024 ** 3 + 128 * 1024 ** 2),
                           quadlets=lambda a: quadlets, runner=lambda a: r, adapter=lambda a: adapter,
                           temp=lambda a: temps['current'], abort=lambda: None, checkpoint=sample_mem,
                           phase=lambda p: None, log=LOGS.append)
    for method in ('addon_member', 'addon_info', 'open'):
        setattr(work, method, MethodType(getattr(haimport.Work, method), work))
    try:
        checked = ha_n8n.check(work, app)
    finally:
        left = conversion.cleanup()
    must('check_temp_objects_removed', not left)
    section = checked['n8n']
    facts['product_check'] = {k: section.get(k) for k in ('owner_email', 'workflows', 'workflows_active',
                                                          'credentials', 'executions', 'users', 'projects',
                                                          'community_packages')}
    facts['product_check_warnings'] = len(checked['warnings'])
    must('product_check_no_blocker', checked['blockers'] == [], blockers=len(checked['blockers']))
    must('product_check_versions_equal', checked['target_version'] == '2.43.6')
    must('product_check_source_facts', section['owner_email'] == OWNER_EMAIL and section['workflows'] == 3
         and section['workflows_active'] == 1 and section['credentials'] == 1 and section['executions'] >= 1
         and section['community_packages'] == [COMMUNITY_PACKAGE]
         and any(w.get('name') == FIX_WORKFLOW for w in section['workflows_to_fix']))

    out['phase'] = 'product-convert'
    shutil.rmtree(staging / 'n8n-work', ignore_errors=True)
    conversion = temps['current'] = tempctr.TempObjects(r, JOB_CONVERT, memory='768m', pids=1024, cpus='1')
    converted_dir = private_dir('work', 'converted')
    try:
        converted = ha_n8n.convert(work, app, False, converted_dir)
    finally:
        left = conversion.cleanup()
        opened.close()
        opened = None
    must('convert_temp_objects_removed', not left)
    must('convert_items', sorted(converted['volumes']) == sorted([DATAVOL, VALKEYVOL])
         and list(converted['postgres']) == ['main'] and converted['owner_email'] == OWNER_EMAIL
         and converted['packages'] == [COMMUNITY_PACKAGE])
    facts['converted'] = {'counts': converted['counts'], 'pg_major': converted['postgres']['main']['major'],
                          'dump_bytes': os.path.getsize(converted['postgres']['main']['file']),
                          'workflows_to_fix': len(converted['workflows_to_fix'])}

    out['phase'] = 'restore'
    safety = private_dir('work', 'safety')
    for volume in VOLUMES:   # harness rollback copy (cold: every container is stopped)

    for volume in (DATAVOL, VALKEYVOL):
        adapter._recreate_volume(volume, converted['volumes'][volume])
    podman('volume', 'rm', '-f', PGVOL, timeout=120)       # like the adapter: the database starts from a new cluster

    start(DB)
    pg = app.postgres[0]
    adapter._wait_for_postgres(pg)
    adapter._check_target_major(pg, {'postgres': [{'id': 'main', 'major': converted['postgres']['main']['major']}]})
    adapter._ensure_database(pg)
    adapter._restore_database(pg, converted['postgres']['main']['file'])
    start(VALKEY)
    start(MAIN)
    verify_restored('installed', creds, converted, target_password)

    out['phase'] = 'restart'
    stop_all()
    for name in (DB, VALKEY, MAIN):
        start(name)
    verify_restored('restarted', creds, converted, target_password)

    out['phase'] = 'rollback'
    stop_all()
    for volume in VOLUMES:
        podman('volume', 'rm', volume, timeout=120)
        podman('volume', 'create', '--label', LABEL, volume, timeout=60)
        import_file(volume, safety / (volume + '.tar'))
    for name in (DB, VALKEY, MAIN):
        start(name)
    t_status, t_cookie = login(TARGET_OWNER, target_password)
    ha_status, _c = login(OWNER_EMAIL, creds['owner_password'])
    rows = {t: sql('SELECT count(*) FROM %s;' % t) for t in ('workflow_entity', 'credentials_entity', '"user"')}
    _m, _u, key_after = tar_member(export_file(DATAVOL, private_dir('work', 'rolled') / 'data.tar'), 'config')
    private(key_after or b'')
    must('rollback_restores_pre_import_target', t_status == 200 and ha_status in (401, 403) and rows == target_rows
         and key_after == target_key, target=t_status, ha=ha_status)
    stop_all()
    out.update(status='passed' if all(checks.values()) else 'partial', phase='done',
               false_checks=sorted(k for k, v in checks.items() if not v))
except Exception as error:  # noqa: BLE001
    out['error_code'] = type(error).__name__
    out['error_where'] = [[Path(f.filename).name, f.lineno, f.name]
                          for f in traceback.extract_tb(error.__traceback__)[-4:]]
    if isinstance(error, AttributeError):
        out['missing_attribute'] = getattr(error, 'name', None)
    if isinstance(error, (AssertionError, KeyError)):
        out['failed_check'] = str(error)[:80]
    if hasattr(error, 'code'):
        out['product_error_code'] = error.code
finally:
    cleanup = {}
    for staged in (KEY_FILE, CREDS_FILE):
        if staged is not None and staged.exists():
            cleanup['staged_file_removed_late'] = True
            staged.unlink()
    if 'opened' in dir() and opened is not None:
        opened.close()
    if 'conversion' in dir() and conversion is not None:
        cleanup['temp_left'] = len(conversion.cleanup())
    for name in (MAIN, VALKEY, DB):
        r.podman('rm', '--force', '--volumes', '--time', '10', name, check=False, timeout=60)
    r.podman('pod', 'rm', '--force', POD, check=False, timeout=120)
    for volume in VOLUMES:
        r.podman('volume', 'rm', '--force', volume, check=False, timeout=60)
    for name in SECRET_NAMES:
        r.podman('secret', 'rm', name, check=False, timeout=60)
    shutil.rmtree(ROOT / 'work', ignore_errors=True)
    shutil.rmtree(ROOT / '.config', ignore_errors=True)
    objects = store_objects(r)
    cleanup.update(store_after={k: len(v) for k, v in objects.items()}, work_removed=not (ROOT / 'work').exists(),
                   no_lab_containers=not objects['containers'], no_lab_pods=not objects['pods'],
                   no_lab_volumes=not objects['volumes'], no_lab_secrets=not objects['secrets'],
                   creds_file_removed=not CREDS_FILE.exists(),
                   unlock_file_removed=KEY_FILE is None or not KEY_FILE.exists())
    out['cleanup'] = cleanup
    if out['status'] == 'passed' and not all(cleanup[k] for k in ('no_lab_containers', 'no_lab_pods', 'no_lab_volumes',
                                                                  'no_lab_secrets', 'creds_file_removed',
                                                                  'unlock_file_removed')):
        out['status'] = 'failed'
        out['failed_check'] = 'cleanup'
    out['elapsed_seconds'] = round(time.monotonic() - started, 1)
    out['podman_commands'] = len(r.commands)

    out['mem_available_min'] = min(MEM) if MEM else None

    needles = {v for v in PRIVATE if len(v) >= 6}

            any(v in commands for v in needles):

    report(ROOT / 'n8n-result.json', out)

    sys.exit(0 if out['status'] == 'passed' else 1)
