import hashlib, json, os, pathlib, tarfile
root = pathlib.Path('/var/tmp/woow-arm54-final-20261123-01xwxjxb'); os.umask(0o077)
def sha(p):
    d = hashlib.sha256()
    with open(p, 'rb') as f:
        for b in iter(lambda: f.read(1 << 20), b''):
            d.update(b)
    return d.hexdigest()
out = {}
for name, want in sorted({'code.tar.gz': 'ed33975590c090178b0c6c7a0910f104c384826f0732fb36340238564bbbf1af', 'client-state.tar': 'b03bb7a62253531395776a7b342ca96be2380b1835d64a78187857f6ff403576', 'ha-backup.tar': '1e6a4665315bc69fd114f8d43f7d429aecab53f07d07793ecf8c2c057a92c5d4'}.items()):
    out[name] = sha(root / 'input' / name) == want
with tarfile.open(root / 'input' / 'code.tar.gz') as tar:
    for m in tar:
        p = pathlib.PurePosixPath(m.name)
        assert m.isfile() and m.size <= 8 * 1024 ** 2 and not p.is_absolute() and '..' not in p.parts
        dest = root / 'source' / m.name
        dest.parent.mkdir(parents=True, exist_ok=True, mode=0o700)
        with dest.open('xb') as handle:
            handle.write(tar.extractfile(m).read())
os.chmod(root / 'input' / 'client-state.tar', 0o600)
out['extracted'] = True
