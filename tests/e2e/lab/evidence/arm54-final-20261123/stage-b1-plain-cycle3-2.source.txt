import hashlib, json, os, pathlib
root = pathlib.Path('/var/tmp/woow-arm54-final-20261123-yy981xjf')
def sha(p):
    d = hashlib.sha256()
    with open(p, 'rb') as f:
        for b in iter(lambda: f.read(1 << 20), b''):
            d.update(b)
    return d.hexdigest()
