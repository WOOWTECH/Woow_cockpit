def private(value):
    if isinstance(value, str):
        value = value.encode()
    if isinstance(value, bytes) and len(value.strip()) >= 6:
        PRIVATE.append(value)
        PRIVATE.append(value.strip())
    return value


def check(name, ok, **detail):
    checks[name] = bool(ok)
    if detail:
        facts[name] = detail
    return bool(ok)


def must(name, ok, **detail):
    if not check(name, ok, **detail):
        raise AssertionError(name)


class Watch(threading.Thread):
    """Every 5 s in .54's own network namespace (started before the engine joins the owner's one): MemAvailable and,
    every 15 s, one GET of .54's own Home Assistant frontend. Below MEM_FLOOR three times in a row the owner's n8n
    target is stopped (the owner's manager is memory capped as well, so .54's own services are never the victims)."""

    def __init__(self):
        super().__init__(daemon=True)
        self.stop = threading.Event()
        self.min_mem, self.samples, self.low, self.tripped = None, 0, 0, False
        self.frontend = {'every_seconds': 15, 'ok': 0, 'failed': 0, 'max_latency_seconds': 0.0, 'over_5s': 0}

    def run(self):
        while not self.stop.is_set():
            for line in Path('/proc/meminfo').read_text().splitlines():
                if line.startswith('MemAvailable:'):
                    value = int(line.split()[1]) * 1024
                    self.min_mem = value if self.min_mem is None else min(self.min_mem, value)
                    self.low = self.low + 1 if value < MEM_FLOOR else 0
            if self.low >= 3 and not self.tripped:
                self.tripped = True
                oc.systemctl('stop', TARGET, timeout=300, check=False)
            if self.samples % 3 == 0:
                t0 = time.monotonic()
                try:
                    with urllib.request.urlopen(FRONTEND, timeout=20) as response:
                        ok = response.status == 200
                except Exception:  # noqa: BLE001 - informational sample
                    ok = False
                took = time.monotonic() - t0
                self.frontend['ok' if ok else 'failed'] += 1
                self.frontend['max_latency_seconds'] = round(max(self.frontend['max_latency_seconds'], took), 3)
                self.frontend['over_5s'] += took > 5
            self.samples += 1
            self.stop.wait(5)


def rid(tag):
    return 'a54n-%s-%s' % (tag, uuid.uuid4().hex[:12])


def call(command, request=None):
    buf = io.StringIO()
    svc = cli.Services(PATHS, launcher=launched.append, unit_active=lambda unit: False)
    code = cli.main([command], stdin=io.StringIO(json.dumps(request or {})), stdout=buf, services=svc)
    reply = json.loads(buf.getvalue().strip().splitlines()[-1])
    if code != 0 or not reply.get('ok'):
        err = reply.get('error') or {}
        facts['cli_refusal'] = {'command': command, 'code': err.get('code'),
                                'message': oc.LONG_TOKEN.sub('<masked>', str(err.get('message')))[:400]}
        raise RuntimeError('cli-refused')
    return reply


def run_job(label, start_command, request, timeout):
    before = len(launched)
    job_id = call(start_command, request)['job_id']
    must('%s_launched_once' % label, launched[before:] == [job_id])
    t0 = time.monotonic()
    p = subprocess.run(['/usr/bin/python3', '-I', '-c', RUN_JOB, job_id], capture_output=True, timeout=timeout,
                       env={'PATH': '/usr/sbin:/usr/bin:/sbin:/bin', 'LANG': 'C.UTF-8'})
    status = call('job-status', {'job_id': job_id})
    job, log = status['job'], status.get('log') or []
    lines = log if isinstance(log, list) else str(log).splitlines()
    jobs_seen[label] = {'job_id': job_id, 'type': job.get('type'), 'state': job.get('state'),
                        'run_job_exit': p.returncode, 'seconds': round(time.monotonic() - t0, 1),
                        'error': (job.get('error') or {}).get('code'),
                        'log_tail': [oc.safe_tail(str(l), 200) for l in lines[-8:]]}
    must('%s_succeeded' % label, job.get('state') == 'succeeded', state=job.get('state'),
         error=jobs_seen[label]['error'], message=oc.safe_tail((job.get('error') or {}).get('message'), 300))
    return job

class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, *args, **kwargs):
        return None


OPENER = urllib.request.build_opener(urllib.request.ProxyHandler({}), NoRedirect())


def http(method, path, body=None, cookie=None, timeout=30):
    """One request to the published loopback port in the owner's private namespace (this thread joined it)."""
    data = json.dumps(body).encode() if body is not None else None
    headers = {'Content-Type': 'application/json', 'Accept': 'application/json', 'browser-id': BROWSER_ID}
    if cookie:
        headers['Cookie'] = 'n8n-auth=' + cookie
    request = urllib.request.Request(BASE_URL + path, data=data, headers=headers, method=method)
    try:
        with OPENER.open(request, timeout=timeout) as response:
            raw = response.read(8 * 1024 * 1024 + 1)
            assert len(raw) <= 8 * 1024 * 1024, 'response-too-large'
            jar = [h for k, h in response.headers.items() if k.lower() == 'set-cookie']
            try:
                return response.status, (json.loads(raw) if raw else None), jar
            except ValueError:
                return response.status, None, jar
    except urllib.error.HTTPError as error:
        return error.code, None, []
    except (urllib.error.URLError, OSError):
        return None, None, []


def wait_ready(seconds=900):
    deadline = time.monotonic() + seconds
    while time.monotonic() < deadline:
        if http('GET', READY, timeout=10)[0] == 200:
            return True
        time.sleep(3)
    raise AssertionError('n8n-not-ready')


def login(email, password):
    status, _body, jar = http('POST', '/rest/login', {'emailOrLdapLoginId': email, 'password': password})
    cookie = None
    for header in jar:
        first = header.split(';', 1)[0]
        if first.startswith('n8n-auth='):
            cookie = first[len('n8n-auth='):]
    if cookie:
        private(cookie)
    return status, cookie


def data_of(body):
    if isinstance(body, dict) and isinstance(body.get('data'), list):
        return body['data']
    if isinstance(body, dict) and isinstance(body.get('data'), dict) and isinstance(body['data'].get('data'), list):
        return body['data']['data']
    return body if isinstance(body, list) else []

def export_file(volume, path):
    with open(path, 'xb') as handle:
        os.fchmod(handle.fileno(), 0o600)
        oc.podman('volume', 'export', volume, stdout=handle, timeout=900)
    return path


def tar_member(path, wanted):
    with tarfile.open(path) as tar:
        for member in tar:
            if member.isfile() and member.name.removeprefix('./') == wanted:
                return member.mode & 0o7777, member.uid, tar.extractfile(member).read()
    return None, None, None


def volume_config(tag):
    """(mode, uid, bytes) of the data volume's `config` file (n8n's encryption key; masked in every output)."""
    path = WORK / ('data-%s.tar' % tag)
    try:
        mode, uid, data = tar_member(export_file(DATAVOL, path), 'config')
    finally:
        path.unlink(missing_ok=True)
    private(data or b'')
    return mode, uid, data


def source_key(keys):
    """HA's n8n encryption key straight from the attested archive (never printed)."""
    with habackup.open_backup(ARCHIVE) as opened:
        member = next(m for m in opened.layout.members
                      if m.role == 'addon' and hastore.addon_suffix(m.slug) == ha_n8n.SLUG)
        with opened.open_member(member.name, keys) as stream:
            select = habackup.Select(include=('config/.n8n/config',), symlinks=(), hardlinks=False)
            for entry in habackup.iter_entries(stream, select, habackup.Report()):
                return json.loads(entry.read_all(65536))['encryptionKey']
    return None


def decrypted_credential(name):
    """n8n's own CLI in the running main container (the REST API redacts secrets); the export exists only inside
    the container and is removed; only found/match leave this function."""
    try:
        oc.podman('exec', MAIN, 'n8n', 'export:credentials', '--all', '--decrypted', '--output=/tmp/a54-creds.json',
                  timeout=600)
        raw = oc.podman('exec', MAIN, 'cat', '/tmp/a54-creds.json', timeout=60).stdout
    finally:
        oc.podman('exec', MAIN, 'rm', '-f', '/tmp/a54-creds.json', check=False, timeout=60)
    rows = json.loads(raw or b'[]')
    row = next((x for x in rows if x.get('name') == name), None)
    value = ((row or {}).get('data') or {}).get('value')
    if isinstance(value, str):
        private(value)
    return row is not None, value


def community_nodes(cookie):
    packages = sql('SELECT "packageName" || \' \' || "installedVersion" FROM installed_packages ORDER BY 1;')
    nodes = [line.split('|') for line in sql('SELECT type, "latestVersion" FROM installed_nodes ORDER BY 1;')
             .splitlines() if line]
    loaded = 0
    for node_type, version in nodes:
        status, body, _jar = http('POST', '/rest/node-types', {'nodeInfos': [{'name': node_type,
                                                                              'version': int(version)}]}, cookie)
        described = data_of(body) if status == 200 else []
        versions = described[0].get('version') if len(described) == 1 else None
        versions = versions if isinstance(versions, list) else [versions]
        if len(described) == 1 and described[0].get('name') in (node_type, node_type.split('.', 1)[-1]) \
                and int(version) in versions:
            loaded += 1
    return packages.splitlines(), len(nodes), loaded

def healthy(container, seconds=300):
    """The package's own HealthCmd, run the way its ExecStartPre does (podman healthcheck run)."""
    deadline = time.monotonic() + seconds
    while time.monotonic() < deadline:
        if oc.podman('healthcheck', 'run', container, check=False, timeout=60).returncode == 0:
            return True
        time.sleep(4)
    return False


def start_target():
    """Like on a device: start the package's target, then wait until every unit runs and n8n answers (a unit whose
    ExecStartPre wait timed out on this slow board is started again by its own Restart=always)."""
    oc.systemctl('start', TARGET, timeout=1800, check=False)
    deadline = time.monotonic() + 1500
    states = {}
    while time.monotonic() < deadline:
        states = {unit: active(unit) for unit in UNITS}
        if all(value == 'active' for value in states.values()) and http('GET', READY, timeout=10)[0] == 200:
            return
        time.sleep(5)
    facts['start_states'] = states
    raise AssertionError('target-not-started')


def stop_target():

    oc.systemctl('stop', *reversed(UNITS), timeout=600, check=False)

def render(data):
    for key, value in RENDER.items():
        data = data.replace('@@%s@@' % key, value)
    assert '@@' not in data, 'unrendered-placeholder'
    return data


def package_files():
    """{relative path: (mode, text)} of the queue-mode installation, and the ExecStartPre lines that were re-pointed."""
    q = PACKAGE / 'quadlet'
    units = {}
    for path in sorted(q.glob('*.container')) + sorted(q.glob('*.volume')) + sorted(q.glob('*.network')):
        units[path.name] = path.read_text()
    units['n8n-runners.container'] = (q / 'optional/n8n-runners.container').read_text()
    for path in sorted((q / 'queue').glob('*.container')) + sorted((q / 'queue').glob('*.volume')):
        units[path.name] = path.read_text()
    units['n8n.container'] += (q / 'queue/n8n-main.fragment').read_text()
    units.pop('n8n-webhook.container')  # N8N_QUEUE_WEBHOOK_PROCESSOR=false
    repointed, files = {}, {}
    for name, text in sorted(units.items()):
        text = render(text)
        count = text.count(PACKAGE_PODMAN)
        if count:
            assert all(PACKAGE_PODMAN not in line or line.startswith('ExecStartPre=') for line in text.splitlines()), \
                'podman-path-outside-ExecStartPre'
            text = text.replace(PACKAGE_PODMAN, HOST_PODMAN)
            repointed[name] = count
        files[QUADLET_DIR + '/' + name] = ('0644', text)
    env = []
    for line in (PACKAGE / 'config/n8n.env.example').read_text().splitlines():
        if line.startswith('EXECUTIONS_MODE='):
            line = 'EXECUTIONS_MODE=queue'
        env.append(line)
    files[CONFIG_DIR + '/n8n.env'] = ('0600', '\n'.join(env) + '\n')
    files[CONFIG_DIR + '/start-valkey.sh'] = ('0644', (PACKAGE / 'config/start-valkey.sh').read_text())
    files[CONFIG_DIR + '/n8n-task-runners.json'] = ('0644', (PACKAGE / 'config/n8n-task-runners.json').read_text())
    files[USER_UNIT_DIR + '/' + TARGET] = ('0644', (PACKAGE / 'systemd' / TARGET).read_text())
    return files, repointed

def verify_restored(phase, creds, section, ha_key, target_password, target_key):
    status, cookie = login(OWNER_EMAIL, creds['owner_password'])
    must(phase + '_ha_owner_logs_in_with_ha_password', status == 200 and bool(cookie), status=status)
    t_status, _t = login(TARGET_OWNER, target_password)
    must(phase + '_target_owner_replaced', t_status in (401, 403), status=t_status)
    s, body, _jar = http('GET', '/rest/workflows', cookie=cookie)
    workflows = [w for w in data_of(body) if isinstance(w, dict)]
    names = sorted(w.get('name') for w in workflows)
    active_count = sum(1 for w in workflows if w.get('active'))
    must(phase + '_workflows', s == 200 and len(workflows) == section['workflows']
         and active_count == section['workflows_active'] and FIX_WORKFLOW in names and MARKER_WORKFLOW not in names,
         workflows=len(workflows), active=active_count)
    s, body, _jar = http('GET', '/rest/credentials', cookie=cookie)
    listed = [c.get('name') for c in data_of(body) if isinstance(c, dict)]
    must(phase + '_credential_listed', s == 200 and CREDENTIAL in listed, credentials=len(listed))
    found, value = decrypted_credential(CREDENTIAL)
    must(phase + '_credential_decrypts_to_ha_value', found and value == creds['credential_value'])
    counts = counts_now()
    must(phase + '_row_counts', counts['workflow_entity'] == section['workflows']
         and counts['credentials_entity'] == section['credentials']
         and counts['execution_entity'] == section['executions'] and counts['"user"'] == section['users']
