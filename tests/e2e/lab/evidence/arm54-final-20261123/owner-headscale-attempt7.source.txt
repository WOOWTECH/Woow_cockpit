def check(name, ok, **detail):
    checks[name] = bool(ok)
    if detail:
        facts[name] = detail
    return bool(ok)


def must(name, ok, **detail):
    if not check(name, ok, **detail):

def rid(tag):

def call(command, request=None):
    buf = io.StringIO()
    svc = cli.Services(PATHS, launcher=launched.append, unit_active=lambda unit: False)
    code = cli.main([command], stdin=io.StringIO(json.dumps(request or {})), stdout=buf, services=svc)
    reply = json.loads(buf.getvalue().strip().splitlines()[-1])
    if code != 0 or not reply.get('ok'):
        err = reply.get('error') or {}
        # Product error messages carry no secret (CONTRACT section 10 rule 4); long key-like runs are masked anyway.
        facts['cli_refusal'] = {'command': command, 'code': err.get('code'),
                                'message': oc.LONG_TOKEN.sub('<masked>', str(err.get('message')))[:400],
                                'details': sorted((err.get('details') or {}).keys())}
        raise RuntimeError('cli-refused')
    return reply


def run_job(label, start_command, request, timeout):
    before = len(launched)
    reply = call(start_command, request)
    job_id = reply['job_id']
    must('%s_launched_once' % label, launched[before:] == [job_id])
    t0 = time.monotonic()
    p = subprocess.run(['/usr/bin/python3', '-I', '-c', RUNNER, job_id], capture_output=True, timeout=timeout,
                       env={'PATH': '/usr/sbin:/usr/bin:/sbin:/bin', 'LANG': 'C.UTF-8'})
    status = call('job-status', {'job_id': job_id})
    job, log = status['job'], status.get('log') or []
    lines = log if isinstance(log, list) else str(log).splitlines()
    jobs[label] = {'job_id': job_id, 'type': job.get('type'), 'state': job.get('state'), 'run_job_exit': p.returncode,
                   'seconds': round(time.monotonic() - t0, 1), 'error': (job.get('error') or {}).get('code'),
                   'log_tail': [oc.safe_tail(str(l), 200) for l in lines[-6:]]}
    (wdir('job-logs') / (label + '.json')).write_text(json.dumps({'job': job, 'log': log}))
    must('%s_succeeded' % label, job.get('state') == 'succeeded', state=job.get('state'),
         error=jobs[label]['error'], message=oc.safe_tail((job.get('error') or {}).get('message'), 300))
    return job.get('result') or {}

def h(value):
    return hashlib.sha256(str(value).encode()).hexdigest()[:16]


NODE_ID_COLS = ('id', 'ipv4', 'ipv6', 'hostname', 'given_name', 'user_id', 'register_method', 'auth_key_id', 'expiry')
NODE_KEY_COLS = ('machine_key', 'node_key')


def db_facts(directory):
    con = sqlite3.connect(str(directory / 'db.sqlite'))  # a private copy; WAL replay needs a writable -shm
    try:
        con.execute('PRAGMA query_only=1')
        integrity = con.execute('PRAGMA integrity_check').fetchone()[0]
        users = sorted(con.execute('select id, name from users').fetchall())
        nodes = []
        for row in con.execute('select %s, %s from nodes order by id' % (', '.join(NODE_ID_COLS), ', '.join(NODE_KEY_COLS))):
            ident = dict(zip(NODE_ID_COLS, row[:len(NODE_ID_COLS)]))
            ident.update({c + '_sha': h(v) for c, v in zip(NODE_KEY_COLS, row[len(NODE_ID_COLS):])})
            nodes.append(ident)
        preauth = sorted((r[0], r[1], r[2], r[3], r[4], h(r[5]), h(r[6]), h(r[7])) for r in con.execute(
            'select id, user_id, reusable, ephemeral, used, prefix, hash, key from pre_auth_keys'))
        api = sorted(h(r[0]) for r in con.execute('select prefix from api_keys'))
        node_keys = {r[0]: r[1] for r in con.execute('select id, node_key from nodes')}
        counts = {t: con.execute('select count(*) from "%s"' % t).fetchone()[0]
                  for t in ('users', 'nodes', 'pre_auth_keys', 'api_keys', 'policies')}
    finally:
        con.close()
    return {'integrity': integrity, 'users': users, 'nodes': nodes, 'pre_auth_keys': preauth,
            'api_key_prefixes_sha': api, 'counts': counts}, node_keys


def unpack_db(tar_path, directory):
    directory.mkdir(mode=0o700, parents=True, exist_ok=True)
    noise = records = None
    names = []
    with tarfile.open(tar_path) as tar:
        for member in tar:
            name = member.name.removeprefix('./')
            names.append(name)
            if member.isfile() and name in ('db.sqlite', 'db.sqlite-wal'):
                (directory / name).write_bytes(tar.extractfile(member).read())
            elif member.isfile() and name == 'noise_private.key':
                noise = tar.extractfile(member).read().strip()
            elif member.isfile() and name == 'extra_records.json':
                records = tar.extractfile(member).read()
    return noise, records, names


def source_member_files(directory):
    directory.mkdir(mode=0o700, parents=True, exist_ok=True)
    wanted = {'data/db.sqlite': 'db.sqlite', 'data/db.sqlite-wal': 'db.sqlite-wal'}
    got = {}
    with habackup.open_backup(ARCHIVE) as opened:
        member = opened.layout.addon('1b7b4ce7_woow-headscale')
        with opened.open_member(member.name, None) as stream:
            select = habackup.Select(include=('data/db.sqlite', 'data/db.sqlite-wal', 'data/noise_private.key',
                                              'config/dns_records.json'), symlinks=(), hardlinks=False)
            for entry in habackup.iter_entries(stream, select, habackup.Report()):
                data = entry.read_all(64 * 1024 ** 2)
                if entry.name in wanted:
                    (directory / wanted[entry.name]).write_bytes(data)
                else:
                    got[entry.name] = data
    return got


def export_volume(volume, dest):
    with open(dest, 'xb') as handle:
        os.fchmod(handle.fileno(), 0o600)
        oc.podman('volume', 'export', volume, stdout=handle, timeout=300)
    return dest

RELAY = r'''import socket, sys, threading
target = (sys.argv[1], int(sys.argv[2]))
def pipe(a, b):
    try:
        while True:
            data = a.recv(65536)
            if not data:
                break
            b.sendall(data)
    except OSError:
        pass
    finally:
        for s in (a, b):
            try:
                s.shutdown(socket.SHUT_RDWR)
            except OSError:
                pass
def serve(ls):
    while True:
        conn, _ = ls.accept()
        try:
            peer = socket.create_connection(target, timeout=10)
        except OSError:
            conn.close()
            continue
        threading.Thread(target=pipe, args=(conn, peer), daemon=True).start()
        threading.Thread(target=pipe, args=(peer, conn), daemon=True).start()
threads = []
for spec in sys.argv[3:]:
    host, port = spec.rsplit(':', 1)
    ls = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    ls.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    ls.bind((host, int(port)))
    ls.listen(64)
    t = threading.Thread(target=serve, args=(ls,), daemon=True)
    t.start()
    threads.append(t)
print('relay-ready', flush=True)
for t in threads:
    t.join()
'''


def rootful(*args, timeout=120, check_rc=True):
    p = subprocess.run(BASE + [str(a) for a in args], env=ENV, capture_output=True, timeout=timeout)
    if check_rc and p.returncode:
        raise RuntimeError('rootful podman %s rc=%d %s' % (args[0], p.returncode, oc.safe_tail(p.stderr)))
    return p


def ts_status():
    p = rootful(*TS, 'status', '--json', timeout=30, check_rc=False)
    try:
        data = json.loads(p.stdout or b'{}')
    except ValueError:
        return {'BackendState': None}
    me = data.get('Self') or {}
    return {'BackendState': data.get('BackendState'), 'AuthURL': bool(data.get('AuthURL')), 'ID': me.get('ID'),
            'HostName': me.get('HostName'), 'TailscaleIPs': me.get('TailscaleIPs'), 'PublicKey': me.get('PublicKey'),
            'Online': me.get('Online')}


def wait_running(seconds):
    seen, state = [], {}
    deadline = time.monotonic() + seconds
    while time.monotonic() < deadline:
        state = ts_status()
        if not seen or seen[-1] != state.get('BackendState'):
            seen.append(state.get('BackendState'))
        if state.get('BackendState') == 'Running' and state.get('TailscaleIPs'):
            break
        time.sleep(2)
    return state, seen

    must('dns_records_are_ha_records', records is not None and records == source_files.get('config/dns_records.json'))
    must('api_keys_are_target_keys_not_ha', rest_db['api_key_prefixes_sha'] == target_db['api_key_prefixes_sha']
         and not set(rest_db['api_key_prefixes_sha']) & set(source_db['api_key_prefixes_sha']))
    must('ha_headplane_api_key_never_read', 'data/.headplane_api_key' in habackup.NEVER_READ)
    plane = export_volume('headplane-data', WORK / 'restored-plane.tar')
    with tarfile.open(plane) as tar:
        marker = {m.name.removeprefix('./'): tar.extractfile(m).read() for m in tar if m.isfile()}
    must('target_headplane_data_kept', marker.get('target-marker.txt') == b'arm54-owner-target-headplane-marker-20261123\n')
    users_after, nodes_after = server_view()
    expected_nodes = {n['id']: sorted([n['ipv4'], n['ipv6']]) for n in source_db['nodes']}
    must('server_lists_ha_users_and_nodes', sorted(users_after.values()) == sorted(n for _i, n in source_db['users'])
         and {k: sorted(v['ips']) for k, v in nodes_after.items()} == expected_nodes)
    api_after = api_probe()
    must('target_api_key_still_works', api_after.get('status') == 200
         and api_after.get('users') == sorted(n for _i, n in source_db['users']))

    out['phase'] = 'real-client'
    for host in ('192.168.2.34', '192.168.2.54'):  # replace: the owner's namespace outlives a failed attempt
        subprocess.run(['ip', 'route', 'replace', 'local', host + '/32', 'dev', 'lo', 'table', 'local'], check=True,
                       capture_output=True, timeout=30)
    relay = subprocess.Popen(['python3', '-I', '-c', RELAY, '127.0.0.1', '28080', '192.168.2.34:28080',
                              '192.168.2.54:28080'], stdout=subprocess.PIPE, stderr=subprocess.DEVNULL,
                             start_new_session=True)
    must('relay_ready', relay.stdout.readline().strip() == b'relay-ready')
    # `ip` (netlink) answers for the joined namespace; /sys/class/net would still show .54's own devices
    if subprocess.run(['ip', 'link', 'show', 'dev', 'eth0'], capture_output=True, timeout=30).returncode != 0:
        subprocess.run(['ip', 'tuntap', 'add', 'dev', 'eth0', 'mode', 'tun'], check=True, capture_output=True,
                       timeout=30)
    for argv in (['ip', 'addr', 'replace', '10.254.54.2/24', 'dev', 'eth0'], ['ip', 'link', 'set', 'eth0', 'up']):
        subprocess.run(argv, check=True, capture_output=True, timeout=30)
    routes = subprocess.run(['ip', 'route'], capture_output=True, text=True, timeout=30).stdout
    must('owner_netns_has_no_default_route', not any(l.startswith('default') for l in routes.splitlines()))
    state_dir, run_dir = wdir('client-state'), wdir('client-run')
    with tarfile.open(ROOT / 'input/client-state.tar') as tar:
        for member in tar:
            p = Path(member.name)
            assert not p.is_absolute() and '..' not in p.parts and (member.isfile() or member.isdir())
            dest = state_dir / p
            if member.isdir():
                dest.mkdir(mode=0o700, parents=True, exist_ok=True)
            else:
                dest.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
                with open(dest, 'xb') as handle:
                    os.fchmod(handle.fileno(), 0o600)
                    handle.write(tar.extractfile(member).read())
    rootful('run', '--detach', '--network', 'ns:/proc/%d/ns/net' % pid, '--name', CLIENT, '--label', LABEL,
            '--pull=never', '--security-opt', 'no-new-privileges', '--memory', '256m', '--pids-limit', '256',
            '--volume', '%s:/var/lib/tailscale' % state_dir, '--volume', '%s:/var/run/tailscale' % run_dir,
            '--entrypoint', 'tailscaled', CLIENT_IMAGE, '--tun=userspace-networking',
            '--state=/var/lib/tailscale/tailscaled.state', '--statedir=/var/lib/tailscale',
            '--socket=/var/run/tailscale/tailscaled.sock', '--no-logs-no-support', '--encrypt-state=false', timeout=180)
    expected_key = source_node_keys.get(2)
    ips = sorted([n for n in source_db['nodes'] if n['id'] == 2][0][k] for k in ('ipv4', 'ipv6'))

    def continuity(name, seconds=120):
        state, seen = wait_running(seconds)
        _u, nodes = server_view()
        node = nodes.get(2) or {}
        must(name, state.get('BackendState') == 'Running' and not state.get('AuthURL') and state.get('ID') == '2'
             and sorted(state.get('TailscaleIPs') or []) == ips and state.get('PublicKey') == expected_key
             and state.get('HostName') == 'b1-ts-alice' and node.get('online') is True
             and len(nodes) == len(source_db['nodes']),
             client={k: v for k, v in state.items() if k != 'PublicKey'}, states=seen, server_node=node)

    continuity('client_keeps_node_identity_on_original_url')
    rootful(*TS, 'down', timeout=60)
    rootful(*TS, 'up', '--login-server', 'http://192.168.2.54:28080', '--hostname', 'b1-ts-alice', timeout=120,
            check_rc=False)
    continuity('client_repointed_to_target_url_keeps_identity')

    out['phase'] = 'restart'
    oc.systemctl('restart', 'headscale.service', timeout=400)
    ok, last = oc.wait_http(pid, HEALTH[0], 180)
    must('healthy_after_owner_unit_restart', ok, last=last)
    continuity('client_reconnects_after_server_restart', 180)
    users_r, nodes_r = server_view()
    must('data_same_after_restart', users_r == users_after and set(nodes_r) == set(nodes_after))

    out['phase'] = 'rollback'
    rootful('stop', '--time', '10', CLIENT, timeout=60, check_rc=False)
    rolled = run_job('rollback', 'restore-start', {'backup_id': safety, 'apps': ['headscale'], 'confirm': True,
                                                    'request_id': rid('rollback')}, 1800)
    facts['rollback'] = {'changed': rolled.get('changed'), 'rolled_back': rolled.get('rolled_back'),
                         'health': [(c.get('id'), c.get('status')) for c in rolled.get('health') or []],
                         'verification': verification_view(rolled),
                         'safety_backup_id': rolled.get('safety_backup_id')}
    must('rollback_restore_ok', rolled.get('changed') is True and rolled.get('rolled_back') is False
         and all(c.get('status') == 'ok' for c in rolled.get('health') or []))
    entries = rolled.get('verification') or []  # the safety backup was made by the product with the entry's digests
    must('rollback_digest_verification', (len(entries) >= 1 and all(e.get('match') is True for e in entries))
         if facts.get('installed_verify') else all(e.get('match') is not False for e in entries))
    users_b, nodes_b = server_view()
    must('target_state_back_after_rollback', sorted(users_b.values()) == ['target-only'] and not nodes_b)
    back = api_probe()
    must('target_api_key_works_after_rollback', back.get('status') == 200 and back.get('users') == ['target-only'])
    export_volume('headplane-data', WORK / 'rollback-plane.tar')
    with tarfile.open(WORK / 'rollback-plane.tar') as tar:
        marker = {m.name.removeprefix('./'): tar.extractfile(m).read() for m in tar if m.isfile()}
    must('target_headplane_marker_after_rollback', marker.get('target-marker.txt') ==
         b'arm54-owner-target-headplane-marker-20261123\n')

    out['phase'] = 'engine-leftovers'
    temp = oc.podman('ps', '-a', '--format', '{{.Names}}').stdout.decode().split()
    must('no_throwaway_containers_left', all(not n.startswith('woowbk-') for n in temp), containers=temp)
    staging = sorted(os.listdir(PATHS.staging)) if PATHS.staging.exists() else []
    must('no_staging_left', staging == [], staging=staging)
    out['status'] = 'passed'
    out['phase'] = 'done'
except Exception as error:  # noqa: BLE001 - phase + type only; secrets never reach the report
    out['error'] = type(error).__name__
    out['error_detail'] = oc.safe_tail(str(error), 600)
finally:
    if relay is not None:
        relay.terminate()
    rootful('rm', '-f', '-t', '5', CLIENT, timeout=60, check_rc=False)
    out['elapsed_seconds'] = round(time.monotonic() - started, 1)
    out['failed_check'] = next((k for k, v in checks.items() if not v), None)
    report(ROOT / 'owner-headscale-result.json', out)
    print(json.dumps(out, sort_keys=True))
sys.exit(0 if out['status'] == 'passed' else 1)
