import hashlib, json, os, pathlib
root = pathlib.Path('/var/tmp/woow-arm54-final-20261123-ukydvpzu')
def sha(p):
    d = hashlib.sha256()
    with open(p, 'rb') as f:
        for b in iter(lambda: f.read(1 << 20), b''):
            d.update(b)
    return d.hexdigest()
out = {name: sha(root / 'input' / name) == want for name, want in sorted({'chain-arm-native.json': 'a47cb15fc85cbcae9e6318bd21f522db904626ed6d7c8a40216b0663928f80b0', 'ha-backup-arm-native.tar': '7fc768a38011acd8681292b62ee9750014ac25137712738442eb616fd1023e29'}.items())}
for name, label in (('ha-key', 'staged_backup_unlock_file_private'), ('n8n-creds.json', 'staged_n8n_login_file_private')):
    path = root / 'input' / name
    out[label] = (not path.exists()) or (path.stat().st_mode & 0o777 == 0o600 and path.stat().st_size > 0)
print(json.dumps(out, sort_keys=True))
assert all(out.values()), 'staged-archive-differs'
